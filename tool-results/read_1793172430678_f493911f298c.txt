    37→type Entry = { expires: number; data: Json };
    38→const mem = new Map<string, Entry>();
    39→
    40→function cacheGet(key: string): Json | null {
    41→  const e = mem.get(key);
    42→  if (!e) return null;
    43→  if (Date.now() > e.expires) {
    44→    mem.delete(key);
    45→    return null;
    46→  }
    47→  return e.data;
    48→}
    49→
    50→function cacheSet(key: string, data: Json, ttlMs: number) {
    51→  mem.set(key, { expires: Date.now() + ttlMs, data });
    52→  if (mem.size > 200) {
    53→    const now = Date.now();
    54→    for (const [k, v] of mem) if (now > v.expires) mem.delete(k);
    55→  }
    56→}

    62→async function upstream(path: string, init?: RequestInit & { timeoutMs?: number }) {
    63→  const timeoutMs = init?.timeoutMs ?? 8000;
    64→  const ctrl = new AbortController();
    65→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    66→  try {
    67→    const res = await fetch(UPSTREAM + path, {
    68→      ...init,
    69→      signal: ctrl.signal,
    70→      headers: { accept: 'application/json', ...(init?.headers || {}) },
    71→    });
    72→    const text = await res.text();
    73→    let data: any = null;
    74→    try {
    75→      data = JSON.parse(text);
    76→    } catch {
    77→      data = null;
    78→    }
    79→    return { ok: res.ok, status: res.status, data };
    80→  } catch {
    81→    return { ok: false, status: 502, data: null as any };
    82→  } finally {
    83→    clearTimeout(timer);
    84→  }
    85→}

    89→function normBiz(it: Json): Json {
    90→  return {
    91→    id: it.id ?? '',
    92→    slug: it.slug ?? '',
    93→    name: it.name ?? '',
    94→    type: it.type ?? '',
    95→    address: it.address ?? '',
    96→    city: it.city ?? '',
    97→    district: it.district ?? '',
    98→    photo: it.photo ?? null,
    99→    about: typeof it.about === 'string' ? it.about : null,
   100→    phone: it.phone ?? '',
   101→    email: it.email ?? '',
   102→    lat: typeof it.lat === 'number' && !isNaN(it.lat) ? it.lat : it.lat != null && !isNaN(Number(it.lat)) ? Number(it.lat) : null,
   103→    lng: typeof it.lng === 'number' && !isNaN(it.lng) ? it.lng : it.lng != null && !isNaN(Number(it.lng)) ? Number(it.lng) : null,

   112→const trLower = (s: unknown) => String(s ?? '').toLocaleLowerCase('tr');
   113→
   114→function filterBiz(items: Json[], q: string, type: string, city: string): Json[] {
   115→  const nq = trLower(q.trim());
   116→  return items.filter((b) => {
   117→    if (type && b.type !== type) return false;
   118→    if (city && b.city !== city) return false;
   119→    if (nq) {
   120→      const svcNames = (b.services as Json[]).map((s) => trLower(s.name)).join(' ');
   121→      const hay = trLower([b.name, b.address, b.district, b.city].join(' ') + ' ' + svcNames);
   122→      if (!hay.includes(nq)) return false;
   123→    }
   124→    return true;
   125→  });
   126→}

   128→async function getBusinesses(): Promise<Json[]> {
   129→  const hit = cacheGet('businesses');
   130→  if (hit && Array.isArray(hit.items)) return hit.items;
   131→  const r = await upstream('/api/public/providers');
   132→  if (!r.ok || !r.data || !Array.isArray(r.data.items)) {
   133→    throw new UpstreamError('İşletme listesi şu anda alınamıyor — ana uygulama yanıt vermiyor olabilir. Birkaç saniye sonra tekrar dene.');
   134→  }
   135→  const items = (r.data.items as Json[]).map(normBiz);
   136→  cacheSet('businesses', { items }, 60_000);
   137→  return items;
   138→}

   140→// 404 öneri sıralaması: arama terimi işletme ADI yanında HİZMET adı/etiketinde de
   141→// aranır; hizmet eşleşmesi en güçlü sinyaldir ("manik" → Manikür hizmetli kuaför).
   142→// Eşleşen hizmet adı _svcMatch alanına yazılır (arayüzde chip olarak gösterilir).
   143→function rankSuggestions(items: Json[], q: string, excludeSlug?: string, limit = 6): Json[] {
   144→  const term = trLower(q).trim().slice(0, 64);
   145→  const pool = excludeSlug ? items.filter((b) => b.slug !== excludeSlug) : items;
   146→  if (!term) return pool.slice(0, limit);
   147→  const scored = pool.map((b) => {
   148→    let score = 0;
   149→    let svcMatch = '';
   150→    const services = Array.isArray(b.services) ? (b.services as Json[]) : [];
   151→    for (const sv of services) {
   152→      const nm = trLower(sv.name);
   153→      const cat = trLower(sv.category);
   154→      if (nm === term) { score = Math.max(score, 100); svcMatch = String(sv.name || ''); break; }
   155→      if (nm.includes(term)) { score = Math.max(score, 60); if (!svcMatch) svcMatch = String(sv.name || ''); }
   156→      else if (cat && (cat === term || cat.includes(term))) { score = Math.max(score, 40); if (!svcMatch) svcMatch = String(sv.category || ''); }
   157→    }
   158→    if (trLower(b.name).includes(term)) score += 30;
   159→    const hay = trLower([b.address, b.district, b.city, b.type].join(' '));
   160→    if (hay.includes(term)) score += 10;
   161→    return { b, score, svcMatch };
   162→  });
   163→  scored.sort((x, y) => y.score - x.score);
   164→  return scored
   165→    .filter((x) => x.score > 0)
   166→    .slice(0, limit)
   167→    .map((x) => (x.svcMatch ? { ...x.b, _svcMatch: x.svcMatch } : x.b));
   168→}

   217→async function renderDetail(slug: string, subdomain: boolean, gwPort?: string | null, q?: string, embed?: boolean): Promise<{ body: string; status: number }> {
   218→  const r = await upstream(`/api/public/providers/${encodeURIComponent(slug)}`);
   219→  if (r.status === 404) {
   220→    // Alt alan adı modunda "tüm işletmelere dön" hedefi: randevu.ANA_DOMAIN ana liste
   221→    const randevuHome = subdomain && BASE_DOMAIN ? `https://${listHost()}` : null;
   222→    // Hata sayfasını zenginleştir: terim hizmet adlarında da aranır (hizmet eşleşmesi öne çıkar)
   223→    let sugg: Json[] = [];
   224→    try {
   225→      const all = await getBusinesses();
   226→      sugg = rankSuggestions(all, q || '', slug);
   227→    } catch {
   228→      // liste alınamadıysa önerisiz 404 gösterilir
   229→    }
   230→    return { body: notFoundPage(slug, gwPort, randevuHome, sugg, subdomain && BASE_DOMAIN ? BASE_DOMAIN : undefined, q), status: 404 };
   231→  }
   232→  if (!r.ok || !r.data || !r.data.provider) {
   233→    throw new UpstreamError('İşletme bilgileri şu anda alınamıyor — ana uygulama yanıt vermiyor olabilir.');
   234→  }
   235→  const services: Json[] = Array.isArray(r.data.services) ? r.data.services : [];
   236→  // vitrin (tur4): portföy fotoğrafları + ekip — upstream hatasına dayanıklı
   237→  const photos: Json[] = Array.isArray(r.data.photos) ? r.data.photos : [];
   238→  const staff: Json[] = Array.isArray(r.data.staff) ? r.data.staff : [];
   239→  const provider = normBiz({ ...r.data.provider, services });
   240→  // Değerlendirmeler (server-side çekilir; hata olursa bölüm boş düşer)
   241→  let reviews: { items: Json[]; summary: Json } | null = null;
   242→  try {
   243→    const rr = await upstream(`/api/public/providers/${encodeURIComponent(slug)}/reviews?limit=20`);
   244→    const rd = rr.data as Json | null;
   245→    if (rr.ok && rd && Array.isArray(rd.items)) {
   246→      reviews = {
   247→        items: rd.items,
   248→        summary: (rd.summary ?? { average: 0, count: 0, distribution: {} }) as Json,
   249→      };
   250→    }
   251→  } catch {
   252→    // değerlendirmeler olmadan da sayfa render edilir
   253→  }
   254→  // Gateway modunda ana sayfa bağlantısına param ekle (CRM'e düşmesin)
   255→  // Düzeltme: "Tüm işletmeler" her zaman randevu LİSTESİNE gider (ana siteye değil)
   256→  const homeUrl = subdomain && BASE_DOMAIN ? `https://${listHost()}` : gwq('/')
   257→  return { body: detailPage({ provider, services, photos, staff, reviews, homeUrl, gwPort, embed }), status: 200 };
   258→
   259→  // BASE_DOMAIN=randevu.x biçimindeyse aynen; bare domain ise randevu. öneki ekle
   260→  function listHost(): string {
   261→    return BASE_DOMAIN.startsWith('randevu.') ? BASE_DOMAIN : `randevu.${BASE_DOMAIN}`;
   262→  }
   263→
   264→  function gwq(p: string) {
   265→    return gwPort ? p + (p.indexOf('?') === -1 ? '?' : '&') + 'XTransformPort=' + encodeURIComponent(gwPort) : p;
   266→  }
   267→}

   271→Bun.serve({
   272→  port: PORT,
   273→  idleTimeout: 30,
   274→  async fetch(req) {
   275→    const url = new URL(req.url);
   276→    const path = url.pathname !== '/' ? url.pathname.replace(/\/+$/, '') : '/';
   277→    const method = req.method.toUpperCase();
   278→
   279→    // Host başlığı → alt alan adı tespiti ({slug}.{BASE_DOMAIN})
   280→    const host = (req.headers.get('host') || '').split(':')[0].trim().toLowerCase().replace(/\.$/, '');
   281→    let subSlug: string | null = null;
   282→    if (BASE_DOMAIN && host.length > BASE_DOMAIN.length + 1 && host.endsWith('.' + BASE_DOMAIN)) {
   283→      const s = host.slice(0, host.length - BASE_DOMAIN.length - 1);
   284→      if (s && !s.includes('.') && !RESERVED_SLUGS.has(s)) subSlug = s;
   285→    }
   286→
   287→    try {
   288→      // Gateway modu tespiti: önizleme paneli ?XTransformPort ile açar;
   289→      // bu durumda göreli varlık/API URL'lerine aynı paramı ekleriz.
   290→      const gwPort = url.searchParams.get('XTransformPort');
   291→
   292→      // --- sağlık kontrolü

   294→

   298→        return htmlRes(page.body, page.status);
   299→      }

   301→      // --- API: işletme listesi (60 sn cache + filtre)

   318→        if (hit) return json(hit);

   320→        if (r.status === 404) return json({ error: 'Bu adda bir işletme bulamadık.' }, 404);
   321→        if (!r.ok || !r.data || !r.data.provider) return json({ error: 'İşletme detayı şu anda alınamıyor. Birazdan tekrar dene.' }, 502);
   322→        const services: Json[] = Array.isArray(r.data.services) ? r.data.services : [];
   323→        const out = { provider: normBiz({ ...r.data.provider, services }), services };

   332→        const date = url.searchParams.get('date') || '';
   333→        const serviceId = url.searchParams.get('serviceId') || '';
   334→        if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) return json({ error: 'Geçerli bir tarih gerekli (YYYY-MM-DD).' }, 400);

   342→        if (r.status === 404) return json({ error: 'İşletme bulunamadı.' }, 404);

   344→        const out = {

   351→          nextAvailableLabel: typeof r.data.nextAvailableLabel === 'string' ? r.data.nextAvailableLabel : null,

   375→        } catch {
   376→          return json({ error: 'İstek gövdesi okunamadı.' }, 400);
   377→        }
   378→        const r = await upstream('/api/public/member/' + rest, {
   379→          method,
   380→          headers: { ...headers, 'content-type': 'application/json' },
   381→          body: JSON.stringify(mbody),
   382→          timeoutMs: 12000,
   383→        });
   384→        if (r.data) return json(r.data, r.ok ? 200 : r.status);
   385→        return json({ error: 'Üyelik servisi şu anda yanıt vermiyor. Birazdan tekrar dene.' }, 502);

   390→      // GET  /api/reviews/providers/<slug>    → GET    /api/public/providers/<slug>/reviews
   391→      if (path === '/api/reviews' && method === 'POST') {
   392→        let rbody: Json;
   393→        try {
   394→          const raw = await req.text();
   395→          rbody = (raw.trim() ? JSON.parse(raw) : {}) as Json;
   396→        } catch {
   397→          return json({ error: 'İstek gövdesi okunamadı.' }, 400);
   398→        }
   399→        const auth = req.headers.get('authorization');
   400→        const headers: Record<string, string> = { 'content-type': 'application/json' };

   408→        if (r.data) return json(r.data, r.ok ? 200 : r.status);

   411→      const mRevList = path.match(/^\/api\/reviews\/providers\/([^/]+)$/);

   426→      // --- API: bekleme listesi proxy'si (CRM /api/public/waitlist → same-origin)
   427→      // POST /api/waitlist → beklemeye gir (üye token'ı varsa Bearer iletilir)

   432→        const auth = req.headers.get('authorization');
   433→        const headers: Record<string, string> = { accept: 'application/json' };
   434→        if (auth) headers['authorization'] = auth;
   435→        const xff = req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip');
   436→        if (xff) headers['x-forwarded-for'] = xff;
   437→        if (method === 'GET') {
   438→          const qs = url.searchParams.toString();
   439→          const r = await upstream('/api/public/waitlist' + (qs ? '?' + qs : ''), { headers });
   440→          if (r.data) return json(r.data, r.ok ? 200 : r.status);

   513→          } catch {
   514→            data = null;
   515→          }

   518→          return json({ error: 'Takvim servisi şu anda yanıt vermiyor. Birazdan tekrar dene.' }, 502);
