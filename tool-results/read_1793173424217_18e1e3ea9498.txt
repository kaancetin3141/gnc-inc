     5→  let ac = null;

     8→  function ctx() {
     9→    if (!ac) ac = new (window.AudioContext || window.webkitAudioContext)();
    10→    if (ac.state === "suspended") ac.resume();
    11→    return ac;
    12→  }
    13→
    14→  function tone(freq, delay, dur, type, vol) {
    15→    const a = ctx();
    16→    const o = a.createOscillator();
    17→    const g = a.createGain();
    18→    o.type = type;
    19→    o.frequency.value = freq;
    20→    const t0 = a.currentTime + delay;
    21→    g.gain.setValueAtTime(vol, t0);
    22→    g.gain.exponentialRampToValueAtTime(0.001, t0 + dur);
    23→    o.connect(g);
    24→    g.connect(a.destination);
    25→    o.start(t0);
    26→    o.stop(t0 + dur + 0.02);
    27→  }
    28→
    29→  return {
    30→    get muted() { return muted; },
    31→    toggle() {
    32→      muted = !muted;
    33→      store.set("cubiq.mute", muted ? "1" : "0");
    34→      return muted;
    35→    },
    36→    pickup() { if (!muted) tone(330, 0, 0.045, "sine", 0.07); },
    37→    place() { if (!muted) { tone(196, 0, 0.07, "triangle", 0.13); tone(262, 0.03, 0.06, "triangle", 0.08); } },
    38→    invalid() { if (!muted) { tone(150, 0, 0.07, "square", 0.05); tone(110, 0.06, 0.09, "square", 0.05); } },
    39→    tick() { if (!muted) tone(880, 0, 0.05, "square", 0.06); },
    40→    clear(lines) {
    41→      if (!muted) {
    42→        const notes = [523, 659, 784, 988, 1175, 1319];
    43→        for (let i = 0; i < Math.min(lines + 1, 6); i++) tone(notes[i], i * 0.07, 0.12, "square", 0.08);
    44→        if (lines >= 2) tone(notes[Math.min(lines, 5)] * 2, lines * 0.07 + 0.05, 0.16, "triangle", 0.07);
    45→        /* hafif yankı */

    78→  function voice(type, freq, start, dur, vol, attack) {
    79→    const a = ctx();
    80→    const o = a.createOscillator();
    81→    const g = a.createGain();
    82→    o.type = type;
    83→    o.frequency.value = freq;
    84→    g.gain.setValueAtTime(0.0001, start);
    85→    g.gain.linearRampToValueAtTime(vol, start + attack);
    86→    g.gain.exponentialRampToValueAtTime(0.0001, start + dur);
    87→    o.connect(g);
    88→    g.connect(a.destination);
    89→    o.start(start);
    90→    o.stop(start + dur + 0.05);
    91→  }

    93→  function playBar(freqs, delay) {
    94→    const a = ctx();
    95→    const start = a.currentTime + delay;
    96→    /* akor pedi (yumuşak) */
    97→    freqs.forEach((f) => voice("sine", f / 2, start, BAR_S + 0.6, 0.024, 0.5));
    98→    /* canlı bas: kök nota pluck'ları (üçüncüde beşlisi) */
    99→    [0, 0.95, 1.9].forEach((t, i) => {
   100→      voice("triangle", (freqs[0] / 2) * (i === 2 ? 1.5 : 1), start + t, 0.3, 0.05, 0.012);
   101→    });
   102→    /* desenli arpej — her barda dönüşümlü figür */
   103→    const pattern = step % 2 === 0 ? [0, 1, 2, 1, 2, 0] : [2, 1, 0, 1, 2, 1];
   104→    pattern.forEach((pi, i) => {
   105→      voice("triangle", freqs[pi] * 2, start + i * (BAR_S / 6), 0.34, 0.026 + (i % 2) * 0.008, 0.02);
   106→    });
   107→    /* tık vurguları */
   108→    [0.6, 1.8].forEach((t) => voice("square", 1568, start + t, 0.04, 0.012, 0.005));
   109→  }
   110→
   111→  function loop() {
   112→    if (!on) return;
   113→    playBar(chords[step % chords.length], 0.06);
   114→    step++;
   115→    timer = setTimeout(loop, BAR_S * 1000);
   116→  }
   117→
   118→  function start() {
   119→    if (!on || timer) return;
   120→    ctx();
   121→    loop();
   122→  }
   123→  function stop() {
   124→    clearTimeout(timer);
   125→    timer = 0;
   126→  }
   127→  return {
   128→    get on() { return on; },
   129→    set on(v) {
   130→      on = !!v;
   131→      DB.set("cubiq.music", on ? "1" : "0");
   132→      if (on) start(); else stop();
   133→    },
   134→    kick: start,
   135→  };
   136→})();

   255→  /* ---------- son 10 oyun sparkline'ı ---------- */
   256→  function buildSpark(hist) {
   257→    if (hist.length < 2) return "";
   258→    const w = 220, h = 36, pad = 3;
   259→    const max = Math.max(...hist, 1);
   260→    const stepX = (w - pad * 2) / (hist.length - 1);
   261→    const pts = hist.map((v, i) => [(pad + i * stepX).toFixed(1), (h - pad - (v / max) * (h - pad * 2)).toFixed(1)]);
   262→    const avg = Math.round(hist.reduce((a, b) => a + b, 0) / hist.length);
   263→    return `<svg class="spark" viewBox="0 0 ${w} ${h}" preserveAspectRatio="none" aria-hidden="true">` +
   264→      `<polyline points="${pts.map((p) => p.join(",")).join(" ")}" fill="none" stroke="#a3eff9" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/>` +
   265→      pts.map((p) => `<circle cx="${p[0]}" cy="${p[1]}" r="2" fill="#ffd93d"/>`).join("") +
   266→      `</svg><div class="sparket">son ${hist.length} oyun · ortalama ${avg}</div>`;
   267→  }

   558→  /* oyun sonu işlemleri — panel görünsün ya da görünmesin TAM BİR KEZ işlenir */
   559→  function applyGameEnd() {
   560→    const st = Game.stats;
   561→    /* kalıcı yaşam boyu istatistikler */
   562→    DB.bumpStats((s) => {
   563→      s.gamesPlayed += 1;
   564→      if (Game.mode === "time") s.gamesTime += 1;
   565→      else s.gamesClassic += 1;
   566→      s.totalLines += st ? st.lines : 0;
   567→      s.totalScore += Game.score;
   568→      if (st) s.bestStreakEver = Math.max(s.bestStreakEver, st.bestStreak);
   569→    });
   570→    /* yıldız sayacı — macerada bölüm yıldızları */
   571→    const stars = Game.mode === "adventure" ? lastAdvStars : earnedStars();
   572→    DB.set("cubiq.stars", String((Number(DB.get("cubiq.stars", 0)) || 0) + stars));
   573→    /* günlük görev sonucu */
   574→    if (Game.mode === "daily") {
   575→      const k = dailyKey();
   576→      const prev = Number(DB.get(k, 0)) || 0;
   577→      if (Game.score > prev) DB.set(k, String(Game.score));
   578→    }
   579→    /* son 10 oyun geçmişi (sparkline) */
   580→    const hist = DB.getJSON("cubiq.history", []);
   581→    hist.push(Game.score);
   582→    while (hist.length > 10) hist.shift();
   583→    DB.setJSON("cubiq.history", hist);
   584→    /* XP: her oyun skor/50 (en az 1); seviye atlanırsa kutlama */
   585→    const gainedXp = Game.score > 0 ? Math.max(1, Math.round(Game.score / 50)) : 0;
   586→    const oldXp = Number(DB.get("cubiq.xp", 0)) || 0;
   587→    const newXp = oldXp + gainedXp;
   588→    DB.set("cubiq.xp", String(newXp));
   589→    if (levelFromXp(newXp) > levelFromXp(oldXp)) {
   590→      Renderer.confetti();
   591→      toast(`🎉 Seviye ${levelFromXp(newXp)}!`);
   592→      Sfx.best();
   593→      vib([15, 30, 15, 30, 45]);
   594→    }
   595→    /* rozetler + liderlik gönderimi */
   596→    checkBadges({ gameScore: Game.score, bestStreak: Game.stats.bestStreak, maxCombo: Game.stats.maxCombo, hintUsed: Game.hintUsed, dailyDays: countDailyDays(), fullClear: lastFullClear });
   597→    /* coin kazanımı: skor/200 + tam temizlik 25 */
   598→    const coinsEarned = Math.round(Game.score / 200) + (lastFullClear ? 25 : 0);
   599→    DB.set("cubiq.coins", String(coins() + coinsEarned));
   600→    lastCoinsEarned = coinsEarned;
   601→    updateCoins();
   602→    renderLeaderboard(Game.mode);
   603→  }

   695→  function trayHit(x, y) {
   696→    for (let i = 0; i < 3; i++) {
   697→      const p = Game.tray[i];
   698→      if (!p) continue;
   699→      const s = Renderer.slotCenter(i);
   700→      const hw = (p.w * Renderer.L.trayCell) / 2 + 16;
   701→      const hh = (p.h * Renderer.L.trayCell) / 2 + 16;
   702→      if (Math.abs(x - s.cx) <= hw && Math.abs(y - s.cy) <= hh) return i;
   703→    }
   704→    return -1;
   705→  }

   707→  function computeTarget() {
   708→    const piece = drag.piece;
   709→    const cx = drag.px, cy = drag.py - drag.lift;
   710→    const Lb = Renderer.L;
   711→    /* hayalet parça merkezi → ızgara hücresi (merkez, hücre merkezine denk gelmeli) */

   713→    const c = Math.round((cx - Lb.boardX) / Lb.cell - piece.w / 2);
   714→    if (Game.canPlace(piece, r, c)) return { r, c, valid: true, sim: Game.previewLines(piece, r, c) };
   715→    return { r, c, valid: false };
   716→  }
